@startuml
title Rétablir une action (v2)
skinparam sequenceArrowColor green
skinparam sequenceLifeLineBorderColor green
skinparam ParticipantBackgroundColor #D5F5D5
skinparam ParticipantBorderColor green

actor User
participant "KeyboardHandle\nConsoleInputHandler" as K
participant EditorController as EC
participant CommandManager as CM
participant EditorEngine as EE
entity selectionx as S
note over S : selectionx:Selection
entity bufferx as B
note over B : bufferx:Buffer
participant EditorMemento as EM
participant "SwingView\nConsoleView" as V

User -> K : Ctrl+Y (ou Ctrl+Maj+Z)
activate K
K -> EC : redo()
deactivate K
activate EC
EC -> CM : redo()
deactivate EC
activate CM

alt redoStack non vide
  CM -> EE : save()
  activate EE
  EE -> EM : new EditorMemento(...)
  activate EM
  EE --> CM : état courant
  CM -> CM : undoStack.push(état courant)
  CM -> CM : redoStack.pop()
  CM -> EE : restore(m)
  deactivate CM
  EE -> EM : text(), start(), end(), anchor()
  deactivate EM
  EE -> B : setText(text)
  activate B
  deactivate B
  EE -> S : restore(start, end, anchor)
  activate S
  deactivate S
  EE -> EE : notifyObservers()
  EE -> V : update()
  deactivate EE
  activate V
  V -> V : render()
  V --> User : éditeur après l'action rétablie
  deactivate V
else redoStack vide
  note over CM : rien ne se passe
end
@enduml
