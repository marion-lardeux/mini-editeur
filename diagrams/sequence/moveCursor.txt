@startuml
title Déplacer le curseur (v1 + v2)
actor User
participant "KeyboardHandle\nConsoleInputHandler" as K
participant EditorController as EC
participant CommandManager as CM #D5F5D5
participant EditorEngine as EE
entity selectionx as S
note over S : selectionx:Selection
entity bufferx as B
note over B : bufferx:Buffer
participant "SwingView\nConsoleView" as V

User -> K : flèche ← ou →
activate K
K -> EC : moveCursor(delta)
deactivate K
activate EC
note right of EC : v1 : EC appelle directement EE.moveCursor(delta)

EC -[#green]> CM : <color:green>execute(cmd, false)</color>
deactivate EC
activate CM
CM -> EE : moveCursor(delta)\n<color:green>(via cmd.execute(), sans save ni empilement)</color>
deactivate CM
activate EE

alt sélection vide
  EE -> S : getStart()
  activate S
  EE <-- S : start
  EE -> B : length()
  activate B
  EE <-- B : longueur
  deactivate B
  note right of EE : pos = clamp(start + delta, 0, longueur)
else sélection non vide
  alt delta < 0
    EE -> S : getStart()
    EE <-- S : start
    note right of EE : pos = start
  else delta > 0
    EE -> S : getEnd()
    EE <-- S : end
    note right of EE : pos = end
  end
end
EE -> S : set(pos, pos)
deactivate S
EE -> EE : notifyObservers()
EE -> V : update()
deactivate EE
activate V
V -> V : render()
User <-- V : éditeur après déplacement
deactivate V
@enduml
