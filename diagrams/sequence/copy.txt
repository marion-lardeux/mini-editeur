@startuml
title Copier du texte (v1 + v2)
actor User
participant "KeyboardHandle\nConsoleInputHandler" as K
participant EditorController as EC
participant CommandManager as CM #D5F5D5
participant EditorEngine as EE
entity clipboardx as C
note over C : clipboardx:Clipboard
entity selectionx as S
note over S : selectionx:Selection
entity bufferx as B
note over B : bufferx:Buffer

User -> K : Ctrl + C
activate K
K -> EC : copy()
deactivate K
activate EC
note right of EC : v1 : EC appelle directement EE.copy()

EC -[#green]> CM : <color:green>execute(cmd, false)</color>
deactivate EC
activate CM
CM -> EE : copy()\n<color:green>(via cmd.execute(), sans save ni empilement)</color>
deactivate CM
activate EE

alt sélection non vide
  EE -> S : getStart()
  activate S
  EE <-- S : start
  EE -> S : getEnd()
  EE <-- S : end
  deactivate S
  EE -> B : substring(start, end)
  activate B
  EE <-- B : texte
  deactivate B
  EE -> C : set(texte)
  note over EE : aucun changement visible : pas de notification
else sélection vide
  note over EE : presse-papier inchangé
end
deactivate EE
@enduml
