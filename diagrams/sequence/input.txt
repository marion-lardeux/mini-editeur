@startuml
title Saisir du texte (v1 + v2)
actor User
participant "KeyboardHandle\nConsoleInputHandler" as K
participant EditorController as EC
participant CommandManager as CM #D5F5D5
participant EditorEngine as EE
entity selectionx as S
note over S : selectionx:Selection
entity bufferx as B
note over B : bufferx:Buffer
participant EditorMemento as EM #D5F5D5
participant "SwingView\nConsoleView" as V

User -> K : touche s
activate K
K -> EC : insert(s)
deactivate K
activate EC
note right of EC : v1 : EC appelle directement EE.insert(s)

EC -[#green]> CM : <color:green>execute(cmd, true)</color>
deactivate EC
activate CM

CM -[#green]> EE : <color:green>save()</color>
activate EE
EE -[#green]> EM : <color:green>new EditorMemento(...)</color>
activate EM
EE -[#green]-> CM : <color:green>before</color>
CM -> EE : insert(s)\n<color:green>(via cmd.execute())</color>

alt s non vide ou sélection non vide
  EE -> EE : replaceSelection(s)
  EE -> S : getStart()
  activate S
  EE <-- S : start
  EE -> S : getEnd()
  EE <-- S : end
  EE -> B : delete(start, end)
  activate B
  EE -> B : insert(start, s)
  deactivate B
  note over EE : pos = start + s.length()
  EE -> S : set(pos, pos)
  deactivate S
  EE -> EE : notifyObservers()
  EE -> V : update()
  activate V
  V -> V : render()
  V -> EE : getText(), getSelStart(), getSelEnd()
  User <-- V : éditeur après saisie
  deactivate V
end

CM -[#green]> EE : <color:green>save()</color>
EE -[#green]-> CM : <color:green>after</color>
deactivate EE
CM -[#green]> EM : <color:green>before.sameText(after)</color>
deactivate EM
alt <color:green>texte modifié</color>
  CM -[#green]> CM : <color:green>undoStack.push(before)</color>
  CM -[#green]> CM : <color:green>redoStack.clear()</color>
end
deactivate CM
@enduml
